    function registrarEntrada() {
      var input = document.getElementById('entradaCodigo');
      var codigo = input.value.trim();
      if (!codigo) return;

      if (_isOfflineReal()) {
        _enfileirarAcao('gravarEntrada', { codigo: codigo, modoAtual: modoEntradaAtual, qtd: 1 });
        showAlert('alertEntrada', '📥 Entrada na fila (offline). Sincronizando...', 'warning');
        input.value = '';
        return;
      }

      load(true);
      google.script.run
        .withSuccessHandler(function(res) {
          load(false);
          if (res.sucesso) {
            showAlert('alertEntrada', res.mensagem + ' (Posição: ' + res.posicao + ')', 'success');
            input.value = '';
            if (!_MODO_SILENCIOSO) {
              _enviarEventoVozMarcado('ENTRADA', 'Entrada confirmada na posição ' + res.posicao, 'painel_gestao');
            }
            carregarPainel();
            carregarUltimasEntradas();
          } else {
            if (res.comando) {
              modoEntradaAtual = res.comando;
              sessionStorage.setItem('modoEntradaDisplayExpress', modoEntradaAtual);
              atualizarModoIndicator();
              showAlert('alertEntrada', res.mensagem, 'info');
              input.value = '';
            } else {
              showAlert('alertEntrada', res.mensagem, 'error');
            }
          }
        })
        .withFailureHandler(function(err) {
          load(false);
          showAlert('alertEntrada', 'Erro de conexão. A entrada foi para a fila offline.', 'warning');
          _enfileirarAcao('gravarEntrada', { codigo: codigo, modoAtual: modoEntradaAtual, qtd: 1 });
          input.value = '';
        })
        .gravarEntrada({ codigo: codigo, modoAtual: modoEntradaAtual, qtd: 1 });
    }

    // MICROFONE (WEB SPEECH API)
    var recognition = null;
    var escutando = false;

    function toggleVozEntrada() {
      var btn = document.getElementById('btnVozEntrada');
      if (!('webkitSpeechRecognition' in window) && !('SpeechRecognition' in window)) {
        alert('Seu navegador não suporta reconhecimento de voz. Use o Chrome ou Edge.');
        return;
      }

      if (escutando) {
        if (recognition) recognition.stop();
        escutando = false;
        btn.style.background = 'rgba(168,85,247,0.2)';
        btn.style.color = '#c4b5fd';
        btn.innerHTML = '&#127908; Falar';
        return;
      }

      var SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
      recognition = new SpeechRecognition();
      recognition.lang = 'pt-BR';
      recognition.continuous = false;
      recognition.interimResults = false;

      recognition.onstart = function() {
        escutando = true;
        btn.style.background = '#ef4444';
        btn.style.color = '#fff';
        btn.innerHTML = '🔴 Ouvindo...';
      };

      recognition.onresult = function(event) {
        var textoFalado = event.results[0][0].transcript;
        var numeroExtraido = extrairNumeroDaFala(textoFalado);
        
        if (numeroExtraido) {
          document.getElementById('entradaCodigo').value = numeroExtraido;
          registrarEntrada();
        } else {
          showAlert('alertEntrada', 'Não entendi o número. Fale "Saca 38", por exemplo.', 'warning');
        }
      };

      recognition.onerror = function(event) {
        if (event.error === 'not-allowed') {
          alert('Você precisa permitir o uso do microfone no navegador (clique no cadeado ao lado do link).');
        }
        toggleVozEntrada();
      };

      recognition.onend = function() {
        escutando = false;
        btn.style.background = 'rgba(168,85,247,0.2)';
        btn.style.color = '#c4b5fd';
        btn.innerHTML = '&#127908; Falar';
      };

      recognition.start();
    }

    function extrairNumeroDaFala(texto) {
      texto = texto.toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '');
      var match = texto.match(/saca\s*(\d+)/i);
      if (match && match[1]) return match[1];
      match = texto.match(/(\d+)/);
      return match ? match[1] : null;
    }

    }
